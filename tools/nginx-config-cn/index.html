SSL / HTTPS

SSLserver { listen 443 ssl http2; server_name example.com; ssl_certificate /etc/nginx/ssl/cert.pem; ssl_certificate_key /etc/nginx/ssl/key.pem; }
HTTPS基本配置
SSLssl_protocols TLSv1.2 TLSv1.3; ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256; ssl_prefer_server_ciphers on; ssl_session_cache shared:SSL:10m; ssl_session_timeout 10m;
SSL协议和密码套件配置
SSLserver { listen 80; server_name example.com; return 301 https://$host$request_uri; }
HTTP自动跳转HTTPS
SSLssl_stapling on; ssl_stapling_verify on; resolver 8.8.8.8 8.8.4.4 valid=300s; resolver_timeout 5s;
启用OCSP Stapling
SSLadd_header Strict-Transport-Security "max-age=63072000; includeSubDomains; preload" always;
启用HSTS头部

负载均衡

负载均衡upstream backend { server 10.0.0.1:8080 weight=3; server 10.0.0.2:8080 weight=2; server 10.0.0.3:8080 backup; }
加权负载均衡
负载均衡upstream backend { least_conn; server 10.0.0.1:8080; server 10.0.0.2:8080; keepalive 32; }
最少连接负载均衡
负载均衡upstream backend { ip_hash; server 10.0.0.1:8080; server 10.0.0.2:8080; }
IP哈希会话保持
负载均衡upstream backend { hash $request_uri consistent; server 10.0.0.1:8080; server 10.0.0.2:8080; }
一致性哈希

安全配置

安全add_header X-Frame-Options "SAMEORIGIN" always; add_header X-Content-Type-Options "nosniff" always; add_header X-XSS-Protection "1; mode=block" always; add_header Referrer-Policy "strict-origin-when-cross-origin" always; add_header Content-Security-Policy "default-src 'self'" always;
安全响应头配置
安全limit_req_zone $binary_remote_addr zone=api:10m rate=10r/s; location /api/ { limit_req zone=api burst=20 nodelay; }
API限流（每秒10请求）
安全location /admin/ { allow 10.0.0.0/8; allow 192.168.1.0/24; deny all; }
IP白名单/黑名单
安全location /admin/ { auth_basic "Restricted"; auth_basic_user_file /etc/nginx/.htpasswd; }
Basic Auth认证
安全if ($request_uri ~* \.(env|git|svn|htaccess)) { return 403; } location ~ /\. { deny all; }
阻止访问敏感文件
安全geo $blocked { default 0; 1.2.3.0/24 1; 5.6.7.0/24 1; } server { if ($blocked) { return 403; } }
