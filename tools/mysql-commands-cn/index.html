DDL - 数据定义

DDLCREATE DATABASE mydb CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
创建数据库
DDLCREATE TABLE users ( id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(100) NOT NULL, email VARCHAR(255) UNIQUE, age INT DEFAULT 0, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ) ENGINE=InnoDB;
创建表
DDLALTER TABLE users ADD COLUMN phone VARCHAR(20); ALTER TABLE users ADD INDEX idx_name (name); ALTER TABLE users ADD CONSTRAINT fk_dept FOREIGN KEY (dept_id) REFERENCES departments(id);
添加列/索引/约束
DDLALTER TABLE users MODIFY COLUMN name VARCHAR(200) NOT NULL; ALTER TABLE users CHANGE COLUMN name full_name VARCHAR(200); ALTER TABLE users DROP COLUMN phone; ALTER TABLE users DROP INDEX idx_name; ALTER TABLE users RENAME TO members;
修改列/重命名列/删除列/删除索引/重命名表
DDLDROP TABLE IF EXISTS users; TRUNCATE TABLE logs; -- 快速清空表数据
删除表 / 清空表
DDLSHOW DATABASES; SHOW TABLES; SHOW COLUMNS FROM users; SHOW CREATE TABLE users; DESC users;
查看数据库/表/表结构

DML - 数据操作

DMLINSERT INTO users (name, email) VALUES ('张三', 'zs@ex.com'); INSERT INTO users (name, email) VALUES ('李四', 'ls@ex.com'), ('王五', 'ww@ex.com');
插入单条/批量插入
DMLINSERT IGNORE INTO users (name, email) VALUES ('张三', 'zs@ex.com'); INSERT INTO users (name, email) VALUES ('赵六', 'zl@ex.com') ON DUPLICATE KEY UPDATE email = VALUES(email);
忽略重复 / 存在时更新
DMLUPDATE users SET age = 26 WHERE id = 1; UPDATE users SET age = age + 1 WHERE dept = 'tech';
更新数据
DMLDELETE FROM users WHERE id = 1; DELETE FROM users WHERE created_at < '2024-01-01' LIMIT 1000;
删除数据（大表建议分批删除）
DMLREPLACE INTO users (id, name, email) VALUES (1, '张三', 'new@ex.com');
存在时删除再插入

DQL - 数据查询

DQLSELECT * FROM users WHERE age > 18 ORDER BY id DESC LIMIT 10 OFFSET 0;
基础查询（条件/排序/分页）
DQLSELECT * FROM users WHERE age BETWEEN 18 AND 30 AND name LIKE '%张%' AND email IS NOT NULL AND dept IN ('tech', 'product');
复杂条件查询
DQLSELECT DISTINCT dept FROM users; SELECT COUNT(DISTINCT dept) FROM users;
去重查询
DQLSELECT name, age, CASE WHEN age < 18 THEN '未成年' WHEN age < 60 THEN '成年' ELSE '老年' END AS age_group FROM users;
条件表达式
DQLSELECT * FROM users WHERE dept_id IN (SELECT id FROM departments WHERE name = 'tech'); SELECT * FROM users u WHERE EXISTS (SELECT 1 FROM orders o WHERE o.user_id = u.id);
子查询（IN / EXISTS）
DQLSELECT name FROM employees UNION SELECT name FROM contractors;
合并查询结果（自动去重）

JOIN 关联查询

DQLSELECT u.name, d.name AS dept FROM users u INNER JOIN departments d ON u.dept_id = d.id;
内连接（两表都有的数据）
DQLSELECT u.name, COALESCE(o.total, 0) AS order_total FROM users u LEFT JOIN (SELECT user_id, SUM(amount) AS total FROM orders GROUP BY user_id) o ON u.id = o.user_id;
左连接（保留左表所有行）
DQLSELECT d.name, u.name AS manager FROM departments d RIGHT JOIN users u ON d.manager_id = u.id;
右连接（保留右表所有行）
DQLSELECT p.name, c.color FROM products p CROSS JOIN colors c;
交叉连接（笛卡尔积）
DQLSELECT e.name AS employee, m.name AS manager FROM users e LEFT JOIN users m ON e.manager_id = m.id;
自连接

聚合与分组

DQLSELECT dept, COUNT(*) AS cnt, AVG(age) AS avg_age FROM users GROUP BY dept HAVING COUNT(*) > 5 ORDER BY cnt DESC;
分组统计（HAVING过滤分组）
DQLCOUNT(*), COUNT(DISTINCT col) SUM(col), AVG(col) MAX(col), MIN(col) GROUP_CONCAT(name SEPARATOR ', ')
常用聚合函数
DQLSELECT name, dept, age, ROW_NUMBER() OVER (PARTITION BY dept ORDER BY age DESC) AS rn, RANK() OVER (ORDER BY age DESC) AS rnk, LAG(age, 1) OVER (ORDER BY id) AS prev_age, SUM(age) OVER (PARTITION BY dept) AS dept_total FROM users;
窗口函数（ROW_NUMBER/RANK/LAG/SUM OVER）
DQLSELECT dept, SUM(salary) AS total FROM employees GROUP BY dept WITH ROLLUP;
分组汇总（含总计行）

索引优化

索引CREATE INDEX idx_name ON users(name); CREATE UNIQUE INDEX idx_email ON users(email); CREATE INDEX idx_composite ON users(dept, age);
创建普通/唯一/复合索引
索引EXPLAIN SELECT * FROM users WHERE name = '张三'; EXPLAIN ANALYZE SELECT * FROM users WHERE age > 18;
分析查询执行计划
索引SHOW INDEX FROM users; SELECT * FROM information_schema.statistics WHERE table_name = 'users';
查看表索引信息
索引-- 最左前缀原则 -- 索引 (a, b, c) 可用于: a / a,b / a,b,c -- 不可用于: b / c / b,c -- 索引失效场景: -- 1. LIKE '%xxx' (左模糊) -- 2. 函数运算: WHERE YEAR(created_at) = 2024 -- 3. 隐式类型转换 -- 4. OR 条件 (改用 UNION) -- 5. NOT IN / !=

事务与锁

事务START TRANSACTION; -- 或 BEGIN UPDATE accounts SET balance = balance - 100 WHERE id = 1; UPDATE accounts SET balance = balance + 100 WHERE id = 2; COMMIT; -- 提交 -- 或 ROLLBACK; -- 回滚
事务基本操作
事务START TRANSACTION; INSERT INTO logs VALUES (1, 'a'); SAVEPOINT sp1; INSERT INTO logs VALUES (2, 'b'); ROLLBACK TO sp1; -- 只回滚第二条 COMMIT;
保存点回滚
事务SELECT * FROM accounts WHERE id = 1 FOR UPDATE; SELECT * FROM accounts WHERE id = 1 LOCK IN SHARE MODE;
行锁（排他/共享）
事务SHOW PROCESSLIST; SHOW ENGINE INNODB STATUS; SELECT * FROM information_schema.innodb_trx;
查看活跃连接和事务
