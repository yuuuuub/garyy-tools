collections - 高级数据结构

数据from collections import defaultdict d = defaultdict(list) d['key'].append(1) # 无需检查key是否存在 d = defaultdict(int) d['count'] += 1 # 默认值为0

数据from collections import deque dq = deque([1, 2, 3]) dq.appendleft(0) # 左端添加 dq.popleft() # 左端弹出 dq.rotate(1) # 右旋转 dq.rotate(-1) # 左旋转 dq = deque(maxlen=5) # 固定长度，自动丢弃旧元素
