📖 读取文件

const fs = require('fs');

// 同步读取
const data = fs.readFileSync('file.txt', 'utf8');
console.log(data);

// 异步读取
fs.readFile('file.txt', 'utf8', (err, data) => {
  if (err) throw err;
  console.log(data);
});

🌊 Stream流

const fs = require('fs');
const { Transform } = require('stream');

// 创建转换流
const upperCase = new Transform({
  transform(chunk, encoding, callback) {
    this.push(chunk.toString().toUpperCase());
    callback();
  }
});

// 使用流处理文件
const readStream = fs.createReadStream('input.txt');
const writeStream = fs.createStream('output.txt');
readStream.pipe(upperCase).pipe(writeStream);

🔧 中间件模式

// 中间件模式实现
function createApp() {
  const middlewares = [];

  function use(fn) {
    middlewares.push(fn);
  }

  async function run(ctx) {
    for (const mw of middlewares) {
      await mw(ctx);
    }
    return ctx;
  }

  return { use, run };
}

const app = createApp();
app.use(async (ctx) => { ctx.user = 'admin'; });
app.use(async (ctx) => { console.log('用户:', ctx.user); });

📡 REST API

const http = require('http');

let todos = [
  { id: 1, text: '学习Node.js', done: false },
  { id: 2, text: '写代码', done: true }
];

const server = http.createServer((req, res) => {
  res.setHeader('Content-Type', 'application/json');

  if (req.url === '/todos' && req.method === 'GET') {
    res.end(JSON.stringify(todos));
  } else if (req.url === '/todos' && req.method === 'POST') {
    let body = '';
    req.on('data', c => body += c);
    req.on('end', () => {
      const todo = JSON.parse(body);
      todo.id = todos.length + 1;
      todos.push(todo);
      res.end(JSON.stringify(todo));
    });
  }
});

server.listen(3000);
