文件系统

import { readTextFile, writeTextFile, ensureDir } from "https://deno.land/std/fs/mod.ts";

// 读取文件
const text = await readTextFile("./data.json");

// 写入文件
await writeTextFile("./output.txt", "Hello");

// 确保目录存在
await ensureDir("./dist");

HTTP 服务器

// 原生 HTTP 服务器
Deno.serve({ port: 8000 }, (req) => {
  const url = new URL(req.url);
  return new Response(`Path: ${url.pathname}`);
});

// Oak 框架
import { Application } from "https://deno.land/x/oak/mod.ts";
const app = new Application();
await app.listen({ port: 8000 });

数据库

import { Database } from "https://deno.land/x/sqlite/mod.ts";

const db = new Database("app.db");
db.execute("CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT)");

db.execute("INSERT INTO users (name) VALUES (?)", ["张三"]);
const users = db.query("SELECT * FROM users");

const { code, stdout } = await cmd.output();

const port = Deno.env.get("PORT") || "8000";

权限检查

// 检查权限
const status = await Deno.permissions.query({ name: "read" });
console.log(status.state); // "granted" | "denied" | "prompt"

// 请求权限
const result = await Deno.permissions.request({ name: "net" });

路径操作

import { join, resolve, basename, extname } from "https://deno.land/std/path/mod.ts";

console.log(join("src", "index.ts"));       // src/index.ts
console.log(resolve("./dist"));              // /abs/path/dist
console.log(basename("/path/file.ts"));      // file.ts
console.log(extname("file.tsx"));            // .tsx

日期格式化

import { format } from "https://deno.land/std/datetime/mod.ts";

const now = new Date();
console.log(format(now, "yyyy-MM-dd HH:mm:ss"));
// 2024-01-15 14:30:00

const uuid = v4.generate();

import { Sha256 } from "https://deno.land/std/hash/sha256.ts";

hash.update("hello world");
