Snowpack 游乐场

配置编辑器
插件示例
常用命令
snowpack.config.js
说明

Svelte 插件

plugins: [
  ['@snowpack/plugin-svelte', {
    input: 'src/App.svelte',
    compilerOptions: {
      dev: true
    }
  }]
]

TypeScript

plugins: [
  ['@snowpack/plugin-typescript', {
    args: '--outDir ./dist',
    watch: 'src/**/*',
  }]
]

PostCSS

plugins: [
  ['@snowpack/plugin-postcss', {
    inputs: ['src/**/*.css'],
    config: {
      path: 'postcss.config.js'
    }
  }]
]

Sass

plugins: [
  ['@snowpack/plugin-sass', {
    native: false
  }]
]

Vite (兼容模式)

plugins: [
  ['@snowpack/plugin-vite', {
    mode: 'development',
    config: {
      server: { port: 3000 }
    }
  }]
]

Bundle Wrapper

plugins: [
  ['snowpack-plugin-bundle', {
    name: 'app.js',
    inputs: ['src/**/*'],
    esmodule: true
  }]
]
npx snowpack dev
启动开发服务器（原生 ESM）
npx snowpack build
生产构建（打包优化）
npx snowpack dev --port 8080
指定端口启动
npx snowpack build --clean
构建前清理输出目录
npx snowpack dev --hot
启用热模块替换
npm install --save-dev snowpack
安装 Snowpack
