build.js (esbuild API)
说明

基础构建

import * as esbuild from 'esbuild';

await esbuild.build({
  entryPoints: ['src/index.ts'],
  bundle: true,
  outfile: 'dist/out.js',
});

开发服务器

import * as esbuild from 'esbuild';

const ctx = await esbuild.context({
  entryPoints: ['src/index.js'],
  bundle: true,
  outdir: 'dist',
  sourcemap: true,
  define: {
    'process.env.NODE_ENV': '"development"'
  }
});

await ctx.watch();
console.log('Watching for changes...');

React + TypeScript

await esbuild.build({
  entryPoints: ['src/index.tsx'],
  bundle: true,
  outfile: 'dist/app.js',
  loader: { '.tsx': 'tsx', '.ts': 'ts' },
  jsx: 'automatic',
  target: 'es2020',
});

CSS 处理

await esbuild.build({
  entryPoints: ['src/styles.css'],
  bundle: true,
  outdir: 'dist',
  loader: {
    '.css': 'local-css',
    '.png': 'file',
    '.svg': 'dataurl'
  },
});

代码分割

await esbuild.build({
  entryPoints: ['src/a.js', 'src/b.js'],
  bundle: true,
  outdir: 'dist',
  splitting: true,
  format: 'esm',
  chunkNames: 'chunks/[name]-[hash]',
});

PostCSS 插件

await esbuild.build({
  entryPoints: ['src/index.js'],
  bundle: true,
  outdir: 'dist',
  plugins: [
    esbuildPostcss({
      plugins: [require('autoprefixer')]
    })
  ],
});
