Bun.serve()

const server = Bun.serve({
  port: 3000,
  fetch(req, server) {
    const ip = server.requestIP(req);
    return new Response(`Hello ${ip?.address}`);
  },
  error(error) {
    return new Response(error.message, { status: 500 });
  },
});

Bun.file()

// 读取文件
const file = Bun.file('./data.json');
const text = await file.text();
const json = await file.json();
const buf = await file.arrayBuffer();

// 检查文件
console.log(file.size);
console.log(file.type);

Bun.write()

// 写入文件
await Bun.write('./output.txt', 'Hello');

// 从 Response 写入
const res = await fetch('https://api.example.com');
await Bun.write('./data.json', res);

// 写入 Buffer
await Bun.write('./binary.bin', buffer);

Bun.spawn()

const proc = Bun.spawn(['ls', '-la']);
const output = await proc.stdout.text();
console.log(output);

// 带输入
const proc2 = Bun.spawn(
  ['grep', 'error'],
  { stdin: new Response('error line') }
);

Bun.sleep()

// 异步等待
await Bun.sleep(1000); // 1秒

// 超时
const result = await Promise.race([
  fetch(url),
  Bun.sleep(5000).then(() => 'timeout')
]);

文件服务器

import { readdir } from 'node:fs/promises';
import { join } from 'node:path';

const dir = process.argv[2] || '.';

Bun.serve({
  port: 3000,
  async fetch(req) {
    const path = new URL(req.url).pathname;
    const file = join(dir, path === '/' ? 'index.html' : path);
    const bunFile = Bun.file(file);
    if (await bunFile.exists()) {
      return new Response(bunFile);
    }
    return new Response('404', { status: 404 });
  },
});

JSON API

let todos = [];

Bun.serve({
  port: 3000,
  async fetch(req) {
    const url = new URL(req.url);
    if (url.pathname === '/todos') {
      if (req.method === 'GET') {
        return Response.json(todos);
      }
      if (req.method === 'POST') {
        const body = await req.json();
        todos.push({ id: Date.now(), ...body });
        return Response.json(todos);
      }
    }
    return new Response('404', { status: 404 });
  },
});

WebSocket 服务器

Bun.serve({
  port: 3000,
  fetch(req, server) {
    if (req.headers.get('upgrade') === 'websocket') {
      return server.upgrade(req);
    }
    return new Response('WebSocket server');
  },
  websocket: {
    open(ws) { ws.send('Connected!'); },
    message(ws, msg) { ws.send(`Echo: ${msg}`); },
    close(ws) { console.log('Client disconnected'); },
  },
});
